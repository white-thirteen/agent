"""Task 1.3.1 real model/app assembly smoke; controlled chat endpoint, no API calls."""
import json
from pathlib import Path
from time import perf_counter
import httpx
from app.core.config import Settings
from app.knowledge.models import HybridSearchResult, SearchQuery
from app.main import create_app
from app.runtime.context import DemoIdentity
from scripts import run_async
async def main():
    settings=Settings(_env_file=None,embedding_enabled=True,knowledge_retrieval_mode='hybrid',
                      model_name='task131-unused',model_api_key='task131-unused',
                      model_base_url='http://127.0.0.1:9',app_env='test')
    app=create_app(settings)
    start=perf_counter()
    identities=DemoIdentity(user_id='dev-user',permission_scopes=frozenset({'knowledge:read'}))
    receipt={}
    async with app.router.lifespan_context(app):
        receipt['start_seconds']=round(perf_counter()-start,3)
        async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app),base_url='http://localhost') as h:
            health=(await h.get('/health')).json()
        assert all(health[x]=='ready' for x in ('database_status','checkpoint_status','store_status','embedding_status'))
        assert health['agent_ready'] and health['knowledge_ready'], health
        leaf=app.state.tool_registry.get('search_knowledge')
        assert leaf.metadata.version=='2'
        receipt.update(health=health,tool_version=leaf.metadata.version,queries=[],paid_api_calls=0)
        print('App resources ready',receipt['start_seconds'],flush=True)
        for query,expected in [('模拟退款会扣真实银行卡的钱吗？','demo-refund-safety'),
                               ('If the stream disconnects, does the accepted task continue?','demo-stream-help')]:
            start=perf_counter()
            result=await app.state.knowledge_retriever.search(SearchQuery(query=query,top_k=3),identity=identities)
            assert isinstance(result,HybridSearchResult) and result.status=='found'
            assert expected in [c.document_id for c in result.evidence]
            assert result.contract_version==2 and result.index_watermark>=2
            refs=await app.state.knowledge_retriever.validate_evidence(result.evidence,identity=identities)
            assert refs['valid']
            receipt['queries'].append({'seconds':round(perf_counter()-start,3),'evidence':[c.document_id for c in result.evidence],
                                       'watermark':result.index_watermark,'source_check':refs,'candidate_count':result.candidate_count})
            print('Hybrid answer evidence',receipt['queries'][-1],flush=True)
    assert app.state.pool.closed and app.state.checkpoint_pool.closed
    receipt['closed']=True
    Path('.git/task131-real-smoke.json').write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
    print('Hybrid resources shut down',flush=True)
run_async(main())
